# Runtime Theme Resource with Type Variations and Light/Dark Switching ## Background Godot 4's GUI skinning system uses the `Theme` resource to drive how every `Control` node is rendered. A theme stores typed items (colors, fonts, font sizes, constants, icons, styleboxes), organized by *theme types* (e.g. `Button`, `Label`, `Panel`). On top of the built-in types, themes can declare **type variations**: named theme types that inherit from a base built-in type and override a subset of items. A `Control` opts into a variation by setting `theme_type_variation` (or by calling `set_theme_type_variation()`), and any unmatched item falls back to the base type. In this task you will build a small Godot 4 project that constructs a `Theme` resource entirely at runtime (not via the inspector and not loaded from a `.tres`), wires it to a root `Control`, exposes a public API to swap between a "light" and a "dark" variant at runtime, and provides two custom button type variations: `PrimaryButton` and `DangerButton`. extends Control

var _current_mode: String = "light"


func _ready() -> void:
	apply_theme("light")


func get_current_mode() -> String:
	return _current_mode


func apply_theme(mode: String) -> void:
	_current_mode = mode
	var t := Theme.new()

	if mode == "dark":
		_build_dark_theme(t)
	else:
		_build_light_theme(t)

	self.theme = t


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------

func _make_flat(color: Color, radius: int) -> StyleBoxFlat:
	var sb := StyleBoxFlat.new()
	sb.bg_color = color
	sb.set_corner_radius_all(radius)
	return sb


func _build_light_theme(t: Theme) -> void:
	# Register type variations so PrimaryButton and DangerButton inherit Button
	t.set_type_variation("PrimaryButton", "Button")
	t.set_type_variation("DangerButton", "Button")

	# Button
	t.set_color("font_color", "Button", Color("#222222"))
	t.set_stylebox("normal",  "Button", _make_flat(Color("#DDDDDD"), 4))
	t.set_stylebox("hover",   "Button", _make_flat(Color("#CCCCCC"), 4))
	t.set_stylebox("pressed", "Button", _make_flat(Color("#AAAAAA"), 4))

	# Label
	t.set_color("font_color", "Label", Color("#222222"))

	# Panel
	t.set_stylebox("panel", "Panel", _make_flat(Color("#FFFFFF"), 0))

	# PrimaryButton
