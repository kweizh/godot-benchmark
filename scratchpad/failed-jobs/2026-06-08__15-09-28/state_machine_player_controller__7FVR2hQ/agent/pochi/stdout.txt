class_name StateMachine
extends Node

## Finite state machine that manages player states.
## Emits state_changed whenever a transition occurs.

signal state_changed(prev: StringName, next: StringName)

## The name of the initial state to enter on _ready.
@export var initial_state: StringName = &"Idle"

var current_state: State = null
var _states: Dictionary = {}


func _ready() -> void:
	# Build state registry from child nodes.
	for child in get_children():
		if child is State:
			_states[StringName(child.name)] = child
			child.player = get_parent() as CharacterBody2D
			child.state_machine = self

	# Enter the initial state without emitting a signal (no previous state).
	if _states.has(initial_state):
		current_state = _states[initial_state]
		current_state.enter()
	else:
		push_warning("StateMachine: initial_state '%s' not found." % initial_state)


func _physics_process(delta: float) -> void:
	if current_state == null:
		return
	var next: StringName = current_state.physics_update(delta)
	if next != &"" and next != StringName(current_state.name):
		transition_to(next)


func _process(delta: float) -> void:
	if current_state != null:
		current_state.update(delta)


## Request a transition to the state identified by `state_name`.
## Transitions to unknown names are silently ignored.
func transition_to(state_name: StringName) -> void:
	if not _states.has(state_name):
		return
	if current_state != null and StringName(current_state.name) == state_name:
		return

	var prev_name: StringName = &""
	if current_state != null:
		prev_name = StringName(current_state.name)
		current_state.exit()

	current_state = _states[state_name]
	current_state.enter()
	state_changed.emit(prev_name, state_name)

## Design decisions

- **Path-based `extends`** (`extends "res://scripts/states/State.gd"`) is used for all concrete states so Godot can resolve the base script during a cold headless parse without depending on the global class registry being fully built first. `class_name` declarations are still present on every file.
- **`StateMachine`** uses duck-typed `Node` references and `has_method()` checks to avoid forward-type resolution issues between `StateMachine.gd` and `State.gd` during scene load.
- **`StateMachine._ready()`** iterates its children, injects `player` and `state_machine` references into every state node, then enters `initial_state` (`Idle`).
- **Transition rules** are implemented entirely inside each state's `physics_update()` return value:
  - `Idle ↔ Run` on horizontal axis input
  - Grounded + `ui_accept` → `Jump` (from Idle or Run)
  - `Jump` → `Fall` when `velocity.y > 0`
  - `Fall` → `Idle` when `is_on_floor()`
  - Any grounded state + `attack` → `Attack`
  - `Attack` → `Idle` after 0.4 s (within the required 0.3–0.6 s window)
- **Unknown transition names** are silently ignored inside `StateMachine.transition_to()`.
- **`state_changed(prev, next)`** is emitted on every transition; `Player._ready()` connects to it and prints state changes.
